# ── SPECIALTIES ──────────────────────────────────────────────
# Crear especialidad
curl -i -X POST "http://localhost:8080/api/specialties" \
  -H "Content-Type: application/json" \
  -d '{"name":"Cardiología"}'

# Listar especialidades
curl -i -X GET "http://localhost:8080/api/specialties"

# Obtener por id
